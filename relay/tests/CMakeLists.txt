cmake_minimum_required(VERSION 3.20)
project(BreathSenseRelayTests C)
enable_testing()
add_executable(relay_core_test test_core.c ../relay_core.c)
target_include_directories(relay_core_test PRIVATE ..)
if(MSVC)
  target_compile_options(relay_core_test PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(relay_core_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME relay_core COMMAND relay_core_test)
set(SILABS_SDK_PATH "" CACHE PATH "Simplicity SDK root (optional BLE event tests)")
if(SILABS_SDK_PATH)
  add_executable(relay_ble_test test_ble.c ../relay_core.c)
  target_include_directories(relay_ble_test PRIVATE stubs .. ../autogen ../config
    "${SILABS_SDK_PATH}/bluetooth_le_host/inc"
    "${SILABS_SDK_PATH}/bgapi_protocol/protocol/inc"
    "${SILABS_SDK_PATH}/platform_common/platform/common/inc")
  if(MSVC)
    target_compile_options(relay_ble_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(relay_ble_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME relay_ble_events COMMAND relay_ble_test)
endif()
